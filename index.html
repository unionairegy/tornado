from pathlib import Path import shutil, zipfile, re, os base = Path("/mnt/data") src = base / "tornado_final_with_phone.html" deploy_dir = base / "tornado_vercel_ready" if deploy_dir.exists(): shutil.rmtree(deploy_dir) deploy_dir.mkdir(parents=True) # Copy the clean HTML as index.html html = src.read_text(encoding="utf-8") # Safety check: make sure no Python script text leaked into the HTML. bad_markers = ["from pathlib import", "html.replace(", "dst.write_text(", "phone_display ="] for marker in bad_markers: if marker in html: raise RuntimeError(f"Unexpected script text found in HTML: {marker}") index_path = deploy_dir / "index.html" index_path.write_text(html, encoding="utf-8") # Copy every image referenced by the page. image_names = re.findall(r']+src="([^"]+)"', html) copied = [] missing = [] for name in sorted(set(image_names)): src_img = base / name if src_img.exists(): shutil.copy2(src_img, deploy_dir / name) copied.append(name) else: missing.append(name) # Create a ZIP ready to upload/deploy. zip_path = base / "tornado_vercel_ready.zip" if zip_path.exists(): zip_path.unlink() with zipfile.ZipFile(zip_path, "w", zipfile.ZIP_DEFLATED) as z: for p in sorted(deploy_dir.iterdir()): z.write(p, arcname=p.name) print("جاهز للرفع على Vercel/GitHub") print("index.html:", index_path) print("ZIP:", zip_path) print("عدد الصور المرفقة:", len(copied)) print("صور ناقصة:", missing if missing else "لا يوجد")
